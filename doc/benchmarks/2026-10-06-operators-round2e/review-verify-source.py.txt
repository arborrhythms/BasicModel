"""Verify saved evidence and package review documents; no training or Git mutation."""
import difflib
import hashlib
import json
import subprocess
import sys
import zipfile
from datetime import datetime, timezone
from pathlib import Path

H = Path(__file__).resolve().parent
ROOT = H.parents[2]
sys.path.insert(0,str(ROOT/'test'))
import bounded_tests
def read(p): return json.loads(p.read_text())
def sha(p): return hashlib.sha256(p.read_bytes()).hexdigest()
def write(p,v): p.write_text(json.dumps(v,indent=2)+'\n')
def git(*args,root=ROOT): return subprocess.check_output(['git',*args],cwd=root,text=True).strip()

source = read(H/'delivered-source/source.json')
assert source == bounded_tests.source_snapshot(ROOT) == read(H/'measurements/source.json')
assert source == read(H/'full-sweep/source-manifest.json')['validated_source']
with zipfile.ZipFile(H/'delivered-source/source.zip') as z:
    assert set(z.namelist()) == set(source)
    assert all(hashlib.sha256(z.read(n)).hexdigest()==v for n,v in source.items())
helpers = read(H/'delivered-source/measurement-helpers.json')
assert all(sha(ROOT/n)==v for n,v in helpers.items())
prior = read(H/'before/intact-receipts.json')
assert all(sha(ROOT/n)==v for n,v in prior.items())
directories = ('2026-10-06-operators-round2','2026-10-06-operators-round2b',
    '2026-10-06-operators-round2c','2026-10-06-operators-round2d','2026-10-06-credit-loop-toy')
current = {str(p.relative_to(ROOT)) for name in directories for p in (H.parent/name).rglob('*')
           if p.is_file() and '__pycache__' not in p.parts}
assert current == set(prior)
untouched = ('doc/Philosophy.md','doc/plans/2026-10-05-operators-update.md')
assert all((ROOT/n).read_bytes()==(H/'before'/f'{n}.txt').read_bytes() for n in untouched)
ports = read(H/'delivered-source/test-ports.json')
with zipfile.ZipFile(H/'before/source.zip') as z:
    for name,body in ports.items():
        before = ((z.read(name).decode() if name in z.namelist() else '') if name.startswith('test/')
                  else (H.parent/'2026-10-06-operators-round2d'/name).read_text())
        after = (ROOT/name if name.startswith('test/') else H/name).read_text()
        assert body == dict(before=before,after=after), name
    for name,body in read(H/'changed-runtime-texts.json').items():
        assert body == dict(old=z.read(name).decode() if name in z.namelist() else '',new=(ROOT/name).read_text())
assert read(H/'delivered-source/seed-port-audit.json')['changed_seed_calls'] == []
inputs = ('measurement-protocol.json','delivered-source/source.json','delivered-source/source.zip',
    'delivered-source/measurement-helpers.json','full-sweep/result.json','full-sweep/initial-result.json',
    'full-sweep/continuation/result.json','full-sweep/coverage-validation.json',
    'measurements/plan.json','measurements/source.json','measurements/complete.json',
    'measurements/sum-read-first.json','measurements/summary.json','measurements/audit-summary.json',
    'mm-first-forward/result.json','paired-mm/result.json','runtime-contract-check.json')
write(H/'measurement-inputs.json',{name:sha(H/name) for name in inputs})
assert read(H/'measurements/complete.json')['completed']
assert read(H/'measurements/complete.json')['retries'] == 0
assert read(H/'full-sweep/coverage-validation.json')['repeated_completed_tests'] == 0
assert read(H/'full-sweep/coverage-validation.json')['failed_reports'] == 0
assert read(H/'results-validation.json')['source_matched']

documents = ('doc/Architecture.md','doc/FutureWork.md','doc/GradientFlow.md',
             'doc/specs/2026-09-29-operator-catalogue.md','todo.md')
patch=[]
for name in documents:
    before=(H/'before'/f'{name}.txt').read_text()
    after=(ROOT/name).read_text()
    patch.extend(difflib.unified_diff(before.splitlines(True),after.splitlines(True),
        fromfile='before-round2e/'+name,tofile='review-round2e/'+name))
(H/'documentation-final.patch').write_text(''.join(patch))
docs=[ROOT/n for n in documents]+sorted(H.glob('*.md'))+sorted(H.glob('reader-*.png'))
with zipfile.ZipFile(H/'documentation-final.zip','w',zipfile.ZIP_DEFLATED) as z:
    for p in docs:z.write(p,str(p.relative_to(ROOT)))
write(H/'documentation-final.json',{str(p.relative_to(ROOT)):sha(p) for p in docs})
supplements=sorted(H.rglob('*.py.txt'))
write(H/'supplemental-helper-hashes.json',{str(p.relative_to(H)):sha(p) for p in supplements})
assert git('rev-parse','HEAD') == '73cd7b71baeb64135c1e2e841b9bfc321339d8bc'
assert git('rev-parse','HEAD',root=ROOT.parent) == 'c9670b545ff88ee1b8176aa44f6c192ad84b597c'
assert git('diff','--cached','--name-only') == ''
assert git('diff','--cached','--name-only',root=ROOT.parent) == ''
state=dict(verified_at=datetime.now(timezone.utc).isoformat(),source_files=len(source),
    source_manifest_sha256=sha(H/'delivered-source/source.json'),source_archive_sha256=sha(H/'delivered-source/source.zip'),
    source_and_archive_matched=True,frozen_helpers=len(helpers),frozen_helpers_matched=True,
    prior_receipt_files=len(prior),prior_receipts_unchanged=True,
    untouched_documents={n:sha(ROOT/n) for n in untouched},complete_test_ports=len(ports),changed_seed_calls=[],
    measurement_inputs_matched=True,supplemental_helpers={str(p.relative_to(H)):sha(p) for p in supplements},
    head=git('rev-parse','HEAD'),parent_head=git('rev-parse','HEAD',root=ROOT.parent),
    index_empty=True,parent_index_empty=True,committed=False,pushed=False,
    standing_gate_passed=read(H/'results-validation.json')['standing_gate_passed'],
    gate_trainings=30,retries=0,replacements=0,
    status=git('status','--short'),parent_status=git('status','--short',root=ROOT.parent))
write(H/'review-state.json',state)
print(json.dumps({k:v for k,v in state.items() if k not in ('status','parent_status','untouched_documents','supplemental_helpers')},indent=2))
