"""Verify and package saved evidence for review; performs no training or Git mutation."""
import difflib, hashlib, json, subprocess, sys, zipfile
from datetime import datetime, timezone
from pathlib import Path

HERE = Path(__file__).resolve().parent
ROOT = HERE.parents[2]
sys.path.insert(0, str(ROOT / 'test'))
import bounded_tests

def read(path): return json.loads(path.read_text())
def sha(path): return hashlib.sha256(path.read_bytes()).hexdigest()
def write(path, value): path.write_text(json.dumps(value, indent=2) + '\n')
def git(*args, root=ROOT):
    return subprocess.check_output(['git', *args], cwd=root, text=True).strip()

source = read(HERE/'delivered-source/source.json')
assert source == bounded_tests.source_snapshot(ROOT)
assert source == read(HERE/'measurements/source.json')
assert source == read(HERE/'full-sweep/source-manifest.json')['validated_source']
with zipfile.ZipFile(HERE/'delivered-source/source.zip') as archive:
    assert set(archive.namelist()) == set(source)
    assert all(hashlib.sha256(archive.read(name)).hexdigest() == digest for name,digest in source.items())
helpers = read(HERE/'delivered-source/measurement-helpers.json')
assert all(sha(ROOT/name) == digest for name,digest in helpers.items())
prior = read(HERE/'before/intact-receipts.json')
assert all(sha(ROOT/name) == digest for name,digest in prior.items())
prior_now = {str(path.relative_to(ROOT)) for directory in ('2026-10-06-operators-round2', '2026-10-06-operators-round2b', '2026-10-06-operators-round2c')
    for path in (HERE.parent/directory).rglob('*') if path.is_file() and '__pycache__' not in path.parts}
assert prior_now == set(prior)
unchanged_documents = ('doc/Philosophy.md', 'doc/plans/2026-10-05-operators-update.md')
assert all((ROOT/name).read_bytes() == (HERE/'before'/f'{name}.txt').read_bytes() for name in unchanged_documents)
assert all(sha(HERE/name) == digest for name,digest in read(HERE/'measurement-inputs.json').items())

ports = read(HERE/'delivered-source/test-ports.json')
with zipfile.ZipFile(HERE/'before/source.zip') as archive:
    for name,text in ports.items():
        before = (archive.read(name).decode() if name in archive.namelist() else '') if name.startswith('test/') else (HERE.parent/'2026-10-06-operators-round2c'/name).read_text()
        after = (ROOT/name if name.startswith('test/') else HERE/name).read_text()
        assert text == dict(before=before,after=after)
assert read(HERE/'delivered-source/seed-port-audit.json')['changed_seed_calls'] == []
runtime_texts = read(HERE/'changed-runtime-texts.json')
with zipfile.ZipFile(HERE/'before/source.zip') as archive:
    for name, text in runtime_texts.items():
        assert text == dict(before=archive.read(name).decode(),after=(ROOT/name).read_text())

documents = ('doc/Architecture.md', 'doc/FutureWork.md', 'doc/GradientFlow.md',
    'doc/specs/2026-09-29-operator-catalogue.md', 'todo.md')
patch = []
for name in documents:
    before = (HERE/'before'/f'{name}.txt').read_text()
    after = (ROOT/name).read_text()
    patch.extend(difflib.unified_diff(before.splitlines(True), after.splitlines(True),
        fromfile='before-round2d/'+name, tofile='review-round2d/'+name))
(HERE/'documentation-final.patch').write_text(''.join(patch))
doc_paths = [ROOT/name for name in documents] + list(HERE.glob('*.md'))
with zipfile.ZipFile(HERE/'documentation-final.zip', 'w', zipfile.ZIP_DEFLATED) as archive:
    for path in doc_paths: archive.write(path, str(path.relative_to(ROOT)))
write(HERE/'documentation-final.json', {str(path.relative_to(ROOT)):sha(path) for path in doc_paths})
supplements = sorted(HERE.glob('*.py.txt'))
write(HERE/'supplemental-helper-hashes.json', {p.name:sha(p) for p in supplements})

assert git('rev-parse','HEAD') == '73cd7b71baeb64135c1e2e841b9bfc321339d8bc'
assert git('rev-parse','HEAD',root=ROOT.parent) == 'c9670b545ff88ee1b8176aa44f6c192ad84b597c'
assert git('diff','--cached','--name-only') == ''
assert git('diff','--cached','--name-only',root=ROOT.parent) == ''
state = dict(verified_at=datetime.now(timezone.utc).isoformat(), source_files=len(source),
    source_manifest_sha256=sha(HERE/'delivered-source/source.json'),
    source_archive_sha256=sha(HERE/'delivered-source/source.zip'),
    source_and_archive_matched=True, frozen_helpers=len(helpers), frozen_helpers_matched=True,
    prior_receipt_files=len(prior), prior_receipts_unchanged=True,
    unchanged_documents={name:sha(ROOT/name) for name in unchanged_documents},
    complete_test_ports=len(ports), changed_seed_calls=[],
    measurement_inputs_matched=True, supplemental_helpers={p.name:sha(p) for p in supplements},
    head=git('rev-parse','HEAD'), parent_head=git('rev-parse','HEAD',root=ROOT.parent),
    index_empty=True, parent_index_empty=True, committed=False, pushed=False,
    status=git('status','--short'),parent_status=git('status','--short',root=ROOT.parent))
write(HERE/'review-state.json',state)
print(json.dumps({k:v for k,v in state.items() if k not in ('status','parent_status','unchanged_documents','supplemental_helpers')},indent=2))
